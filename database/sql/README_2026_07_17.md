## 🧪 Verification Queries

Run these queries untuk memastikan semua perubahan berhasil:

```sql
-- Check report_categories
SELECT id, slug, name, sort_order, is_active FROM report_categories ORDER BY sort_order;

-- Check hero_slider_settings
SELECT
    min_width, min_height, max_width, max_height,
    max_file_size_mb, aspect_ratio, slider_delay_ms,
    enable_autoplay, enable_touch_swipe,
    enable_navigation_arrows, enable_dot_indicators
FROM hero_slider_settings;

-- Check admin menu
SELECT id, key, name, route, section, `order` FROM admin_menus WHERE key = 'hero-slider-settings';

-- Check permissions count
SELECT COUNT(*) as total_permissions FROM admin_menu_permissions amp
JOIN admin_menus am ON am.id = amp.admin_menu_id
WHERE am.key = 'hero-slider-settings';

-- Detailed permissions
SELECT
    am.name as menu_name,
    r.name as role_name,
    amp.can_access,
    amp.created_at
FROM admin_menu_permissions amp
JOIN admin_menus am ON am.id = amp.admin_menu_id
JOIN roles r ON r.id = amp.role_id
WHERE am.key = 'hero-slider-settings'
ORDER BY r.name;
```
